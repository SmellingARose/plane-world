# pwgpu: thin GPU backend abstraction. Metal (Objective-C++, ARC) on Apple, Direct3D 12 on Windows.
add_library(pwgpu STATIC src/device.cpp)
target_include_directories(pwgpu PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(pwgpu PUBLIC pw_flags PRIVATE pw_shaders)
if(APPLE)
  target_sources(pwgpu PRIVATE src/metal_device.mm)
  set_source_files_properties(src/metal_device.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
  target_link_libraries(pwgpu PUBLIC "-framework Metal" "-framework Foundation")
  target_compile_definitions(pwgpu PUBLIC PW_GPU_METAL=1)
elseif(WIN32)
  target_sources(pwgpu PRIVATE src/d3d12_device.cpp)
  target_link_libraries(pwgpu PUBLIC d3d12 dxgi dxguid)
  target_compile_definitions(pwgpu PUBLIC PW_GPU_D3D12=1)
else()
  message(WARNING "pwgpu: no GPU backend for this platform (Metal and Direct3D 12 only)")
endif()
